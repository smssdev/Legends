lib/model.mjs
// نموذج الكتاب: Qwen3 4B Instruct (2507)، نموذج مفتوح برخصة Apache 2.0،
// ضغطته Unsloth إلى 4 بتات (2.5 GB). يُنزَّل مرة، أول مرة يعمل فيها مثال،
// في ~/.node-llama-cpp/models؛ ثم يعمل كل شيء على حاسوبك، دون اتصال.
import {getLlama, LlamaLogLevel, resolveModelFile} from "node-llama-cpp";

export const MODEL = "hf:unsloth/Qwen3-4B-Instruct-2507-GGUF:Q4_K_M";

export async function loadModel() {
  const llama = await getLlama({gpu: false, logLevel: LlamaLogLevel.error});
  return llama.loadModel({modelPath: await resolveModelFile(MODEL)});
}

تنزّل resolveModelFile النموذج من Hugging Face، وهي مكتبة عامة للنماذج المفتوحة، في المرة الأولى، وتجده على القرص بعد ذلك. وgpu: false تشغّله على المعالج، فلا تتوقف أرقام الكتاب على بطاقة رسوميات؛ وإن كانت لديك واحدة، فحذف هذا الخيار يجعل كل شيء أسرع ولا يغيّر شيئاً آخر.

1.3 الرموز

النموذج لا يقرأ حروفاً ولا كلمات. فمُرمِّزه يقطّع النص إلى رموز من مفردات ثابتة: الكلمات الشائعة رمز واحد، والنادرة عدة رموز، وأي شيء يمكن دائماً تهجئته قطعة قطعة. وهذا البرنامج يقطّع الجملة نفسها بالإنجليزية وبالعربية:

tokens.mjs
import {loadModel} from "./lib/model.mjs";

const model = await loadModel();

// الجملة نفسها بلغتين، وكيف يقطّع النموذج كلاً منهما إلى رموز.
const sentences = [
  "The café opens at 8:30 and closes at 22:00.",
  "يفتح المقهى في 8:30 ويغلق في 22:00.",
];
for (const sentence of sentences) {
  const tokens = model.tokenize(sentence);
  const pieces = tokens.map((token) => model.detokenize([token]));
  const words = sentence.split(" ").length;
  const size = `${sentence.length} characters, ${tokens.length} tokens`;
  console.log(`${words} words, ${size}`);
  console.log("  " + pieces.map((piece) => `[${piece}]`).join(""));
}

وهذا مهم لأن الرموز وحدة كل شيء. فالنماذج المستضافة تُسعَّر بالرمز؛ ونافذة السياق، أي مقدار النص الذي يستطيع النموذج النظر فيه دفعة واحدة، تُقاس بالرموز؛ والسرعة رموز في الثانية. والمشروع 1.1 يقيس الفرق على جملة كاملة: 1.18 رمز لكل كلمة إنجليزية، و2.63 لكل كلمة عربية. فالمحادثة نفسها بالعربية تكلّف أكثر، وتملأ النافذة أسرع، وتأخذ وقتاً أطول، مع هذا النموذج. ولنماذج أخرى مفردات أخرى، فقِس ما تستخدمه.

انتبه

عدّ الحروف، وقلب الكلمات، والحساب على الأرقام الطويلة صعبة على النماذج للسبب الذي رأيته للتو: النموذج لا يرى الحروف أبداً، بل القطع فقط. وحين تحتاج عدّاً دقيقاً أو ترتيباً أو مجموعاً، اطلب من النموذج أن يكتب كوداً يفعل ذلك، وشغّل الكود.

اختبر نفسك

  1. لماذا أعلى احتمال في الجدول العربي 7.3% فقط، بينما هو 40% في الإنجليزي؟
  2. نص فيه 1,000 كلمة إنجليزية. كم رمزاً تقريباً عند هذا النموذج، وكم لو كان 1,000 كلمة عربية؟
  3. لماذا سؤال «كم حرفاً في هذه الكلمة؟» أصعب على النموذج من «ما معنى هذه الكلمة؟»

الإجابات في آخر الفصل، بعد الملخص.

1.4 درجة الحرارة

الاحتمالات ثابتة؛ أما كيف يُختار الرمز التالي منها فقرارك. في درجة الحرارة 0 يفوز الرمز الأرجح دائماً. والدرجات الأعلى تسطّح الاحتمالات ثم تسحب عشوائياً، فتأخذ الرموز الأقل احتمالاً فرصة. والبذرة تثبّت السحب العشوائي، فالبذرة نفسها تكرر الاختيارات نفسها:

temperature.mjs
import {LlamaCompletion} from "node-llama-cpp";
import {loadModel} from "./lib/model.mjs";

const model = await loadModel();
const context = await model.createContext({contextSize: 512});
const completion = new LlamaCompletion({
  contextSequence: context.getSequence(),
});

// درجة الحرارة 0 تأخذ دائماً الرمز الأرجح؛ والدرجات الأعلى تسحب عشوائياً،
// فيتغير الجواب مع البذرة.
const request = "اسم قصير ولطيف لمقهى جديد:";
for (const temperature of [0, 1.2]) {
  console.log(`temperature ${temperature}:`);
  for (const seed of [1, 2, 3]) {
    const text = await completion.generateCompletion(request, {
      temperature,
      seed,
      maxTokens: 12,
    });
    console.log(`  seed ${seed}: ${text.trim().split("\n")[0]}`);
  }
}

1.6 واثق ومخطئ

بحث «تخمير الرموز» لليلى حسن غير موجود؛ اختُلق لهذا القسم. وهذا البرنامج يطلب من النموذج تلخيصه، مرة بسؤال عادي، ومرة مع السماح له بأن يقول إنه غير متأكد:

invent.mjs
import {LlamaChatSession} from "node-llama-cpp";
import {loadModel} from "./lib/model.mjs";

const model = await loadModel();
const context = await model.createContext({contextSize: 2048});

const ask =
  "لخّص في جملتين بحث «تخمير الرموز» " +
  "الذي نشرته ليلى حسن عام 2019.";
const doubt = "وإن لم تكن متأكداً من وجود البحث، فقل ذلك.";
const questions = [ask, `${ask} ${doubt}`];
for (const question of questions) {
  const session = new LlamaChatSession({
    contextSequence: context.getSequence(),
    autoDisposeSequence: true,
  });
  const answer = await session.prompt(question, {
    temperature: 0,
    maxTokens: 120,
  });
  console.log(`> ${question}\n${answer}\n`);
  session.dispose();
}

في المرة الأولى كتب النموذج ملخصاً طليقاً في جملتين لنتائج بحث لا وجود له، عن كيف «تُخمَّر» الرموز عبر السياق والتجربة الشخصية، ولا شيء فيه يدل على أنه مختلَق. وفي المرة الثانية، حين سُمح له بالشك، قال الحقيقة. وفي النسخة الإنجليزية من الكتاب، طُرح السؤال نفسه بالإنجليزية على النموذج نفسه بالدرجة نفسها، فرفض في المرتين: «لا يوجد بحث معروف بهذا العنوان».

هذا ما يسميه الناس الهلوسة، وقد شرحه القسم 1.2 من قبل. فالنموذج لا يبحث عن شيء. إنه ينتج التكملة الأرجح للنص، والتكملة الأرجح لـ «لخّص هذا البحث» ملخص. وهل يرفض أم لا يتوقف على ما إذا كان الرفض شائعاً، في تدريبه، بعد نص يشبه نصك: بالإنجليزية، حيث رأى أمثلة أكثر على «لا يوجد بحث كهذا»، رفض؛ وبالعربية لم يرفض. ولم يعطِ النموذج أي علامة على أي الجوابين صحيح. ولا يعطي أبداً: فالجواب الخاطئ يخرج بالطلاقة نفسها التي يخرج بها الصحيح.

انتبه

أخطر الاختلاقات المعقولة منها: دالة يبدو أنها تنتمي إلى مكتبة، وخيار إعدادات بأسلوب التسمية الصحيح، ومرجع لمصدر يبدو حقيقياً. وقبل أن تعتمد على أي اسم أو رقم أو اقتباس أو رابط من نموذج، تحقق منه حيث يُعرَّف: التوثيق، أو الكود، أو المصدر نفسه. وتحقق مرتين حين تسأل بالعربية: فقد رأيت للتو أن النموذج نفسه أصدق بالإنجليزية.

والنماذج المستضافة الأكبر تختلق أقل، وكثير منها يستطيع اليوم البحث في الويب أو قراءة ملفاتك، وهذا يساعد كثيراً. لكن لا شيء من ذلك يغيّر الآلية. وفي هذا الكتاب كله ستستخدم ثلاثة دفاعات، وقد أظهر هذا القسم اثنين منها: أعطِ النموذج الحقائق بدل أن تطلب منه تذكّرها (الفصل 14)، واسمح بـ «لا أعرف» واطلبها (الفصل 3)، وتحقق مما يهم (كل فصل).

  • النموذج اللغوي يفعل شيئاً واحداً: بعد الرموز حتى الآن، يعطي احتمالاً لكل رمز ممكن بعدها. والتوليد يكرر ذلك، رمزاً رمزاً.

  • المعاملات تحمل ما تعلّمه النموذج، على هيئة أنماط. ومعرفته تتوقف عند تاريخ تدريبه، ولا يتذكر شيئاً بين المحادثات.
  • النماذج تختلق أجوبة طليقة حين لا تكون التكملة الأرجح هي الصحيحة. أعطها الحقائق، واسمح لها بقول «لا أعرف»، وتحقق مما يهم.

راقب الحلقة

دع النموذج يكمل جملة ثمانية رموز، واطبع كل اختيار مع الاحتمال الذي كان له.

ستتدرب على: تقييم تسلسل رمزاً رمزاً، وقراءة ثقة النموذج.

  1. رمّز الأمر ومرّره إلى evaluateWithMetadata مع confidence: true.
  2. كل قيمة تعطيك إياها الحلقة هي الرمز المختار التالي؛ اطبعه واطبع احتماله.
  3. توقف بعد ثماني خطوات.

import {loadModel} from "./lib/model.mjs";

const model = await loadModel();
const context = await model.createContext({contextSize: 512});
const sequence = context.getSequence();

// التوليد حلقة: تنبّأ، واختر، وأضف، وكرر. وهنا يُعرض كل اختيار مع
// احتماله.
const prompt = model.tokenize("المقهى يبيع القهوة و");
let step = 0;
const options = {confidence: true};
for await (const output of sequence.evaluateWithMetadata(prompt, options)) {
  const piece = JSON.stringify(model.detokenize([output.token])).padEnd(14);
  const percent = (output.confidence * 100).toFixed(1);
  console.log(`step ${++step}: ${piece} ${percent}%`);
  if (step === 8) {
    break;
  }
}

2.2 المحادثة: المحادثة كلها، في كل مرة

جلسة المحادثة في node-llama-cpp تحفظ المحادثة لك. وهذا البرنامج يسأل سؤالين في جلسة واحدة، ويطبع كم رمزاً في السياق بعد كل جواب، ثم يسأل السؤال الثاني من جديد في جلسة جديدة:

chat.mjs
import {LlamaChatSession} from "node-llama-cpp";
import {loadModel} from "./lib/model.mjs";

const model = await loadModel();
const context = await model.createContext({contextSize: 2048});
const options = {temperature: 0, maxTokens: 120};
const systemPrompt = "أنت مساعد موجز لمطوري JavaScript. أجب في سطرين على الأكثر.";

// النموذج لا يتذكر شيئاً: في كل دور تُرسل المحادثة كلها من جديد.
const sequence = context.getSequence();
const session = new LlamaChatSession({contextSequence: sequence, systemPrompt});
for (const question of ["كيف أقرأ ملف JSON في Node؟", "وكيف أكتب واحداً؟"]) {
  const answer = await session.prompt(question, options);
  // كم رمزاً تأخذ المحادثة الآن في السياق.
  console.log(`> ${question}\n${answer}\n[${sequence.nextTokenIndex} tokens]\n`);
}

sequence.dispose();
console.log("محادثة جديدة، سُئلت السؤال الثاني وحده:");
const fresh = new LlamaChatSession({
  contextSequence: context.getSequence(),
  systemPrompt,
});
console.log(`> وكيف أكتب واحداً؟\n${await fresh.prompt("وكيف أكتب واحداً؟", options)}`);

كل منتجات المحادثة تعمل هكذا. النموذج لا يتذكر شيئاً؛ والتطبيق يرسل المحادثة من جديد مع كل رسالة. وتتبع ذلك ثلاث قواعد عملية:

  • المحادثة الطويلة تبطؤ، ومع النماذج المستضافة تغلو، مع كل رسالة، لأن كل رسالة تعيد إرسال كل ما قبلها.
  • حين تتجاوز المحادثة نافذة السياق، يضطر التطبيق إلى حذف أقدم الرسائل أو تلخيصها، فينسى النموذج ما قلته في البداية.
  • حين يبدأ المساعد بخلط مهمة قديمة بجديدة، ابدأ محادثة جديدة. ليس هذا قلة ذوق؛ إنه سياق نظيف.

نصيحة

أمر النظام هو التعليمات الموضوعة قبل المحادثة: هنا «مساعد موجز لمطوري JavaScript، في سطرين على الأكثر». ومنتجات المحادثة تتيح لك ضبطه، وكثيراً ما يسمى «تعليمات مخصصة» أو «أمر المشروع». ضع فيه ما يصدق على كل سؤال: لغتك، وإطار عملك وإصداراته، وطول الأجوبة. فلا تكرره في كل رسالة.

2.3 الإكمال: الكود حول مؤشرك

مساعد المحرر لا يحاور. فأثناء كتابتك، يرسل الكود الذي قبل المؤشر والكود الذي بعده، ويطلب من النموذج القطعة التي بينهما. وهذا يسمى ملء الوسط، وبعض النماذج، ومنها نموذج الكتاب، دُرّبت عليه:

complete.mjs
import {LlamaCompletion} from "node-llama-cpp";
import {loadModel} from "./lib/model.mjs";

const model = await loadModel();
const context = await model.createContext({contextSize: 1024});
const completion = new LlamaCompletion({
  contextSequence: context.getSequence(),
});

// المحرر يرسل الكود قبل المؤشر وبعده، ويطلب ما بينهما.
const before = `// Total price of an order, with tax.
export function total(items, taxRate) {
`;
const after = `
}

export const TAX_RATE = 0.14;
`;

// بلا قيد، يستمر النموذج في الكتابة بعد الدالة؛ والمحرر يتوقف عند نهاية
// الكتلة.
for (const stop of [[], ["\n}"]]) {
  const middle = await completion.generateInfillCompletion(before, after, {
    temperature: 0,
    maxTokens: 60,
    customStopTriggers: stop,
  });
  console.log(stop.length ? "--- stopped at the block's end:" : "--- raw:");
  console.log(middle.replace(/^\n/, ""));
}

ولاحظ أيضاً ما افترضه النموذج. التعليق يقول «مجموع سعر الطلب» لا أكثر، فاخترع شكل العنصر: item.price وitem.quantity. فإن كان في عناصرك count، فالاقتراح خاطئ، وهو يبدو تماماً كالكود الصحيح. فالإكمال لا يعرف إلا ما قرب المؤشر، وما اختار المحرر إرساله من ملفات أخرى. افتح الملفات التي يجب أن يعرفها، وسمِّ الأشياء بوضوح: كلما وضح الكود حول المؤشر، تحسّن الاقتراح.

على حاسوب الكتاب، معالج بأربع أنوية بلا بطاقة رسوميات، يأتي أول رمز في نحو ربع ثانية، ثم تتبعه ثمانية رموز في الثانية: جواب من سطرين في ثوانٍ، وصفحة كاملة في دقيقتين تقريباً. وهذا يكفي للتعلم ولتشغيل المهام الصغيرة، ولهذا تطلب أمثلة الكتاب أجوبة قصيرة. والنماذج المستضافة تعمل على صفوف من المسرّعات، وهي عادة أسرع بأضعاف، وبطاقة الرسوميات تسرّع النموذج المحلي كثيراً. وفي إكمال المحرر السرعة كل شيء: الاقتراح الذي يصل بعد أن كتبت السطر بلا فائدة، ولهذا تستخدم أدوات المحرر نماذج صغيرة سريعة وسياقاً قليلاً.

2.6 ما تستطيع الأداة رؤيته

المساعد لا يستطيع أن يستخدم إلا ما في سياقه. فكم هذا، مقابل قاعدة كود حقيقية؟ وهذا البرنامج يعدّ الرموز في مكتبة واحدة، node-llama-cpp نفسها:

context.mjs (جزء)
// كم نصاً في مكتبة حقيقية، مقابل حجم نافذة السياق.
const folder = "node_modules/node-llama-cpp/dist";
let files = 0;
let tokens = 0;
for (const name of fs.readdirSync(folder, {recursive: true})) {
  if (name.endsWith(".js")) {
    const text = fs.readFileSync(path.join(folder, name), "utf8");
    files++;
    tokens += model.tokenize(text).length;
  }
}
console.log(`node-llama-cpp: ${files} JavaScript files, ${tokens} tokens`);
for (const window of [8_192, 128_000, 1_000_000]) {
  const share = Math.min(100, (100 * window) / tokens).toFixed(0);
  console.log(`a ${window}-token window holds ${share}% of it`);
}

عادة المحترفين

أبقِ الأسرار خارج الملفات التي يستطيع المساعد رؤيتها: في متغيرات البيئة أو مدير أسرار، لا في الكود أبداً، ولا في ملف تلصقه في محادثة. ومعظم الأدوات تتيح أيضاً استثناء ملفات أو مجلدات؛ فاستثنِ كل ما فيه مفاتيح أو بيانات عملاء أو إعدادات بيئة الإنتاج.

  • كل أداة ذكاء اصطناعي نموذج في واحد من ثلاثة أشكال: محادثة، أو إكمال، أو وكيل بأدوات.

  used += fits ? tokens : 0;

}

const calculate = defineChatSessionFunction({
  description: "Evaluate an arithmetic expression, such as (2 + 3) * 4",
  params: {type: "object", properties: {expression: {type: "string"}}},
  handler({expression}) {
    if (!/^[\d\s.+\-*/()]+$/.test(expression)) {
      return "error: only numbers and + - * / ( )";
    }
    const result = Function(`return (${expression});`)();
    console.log(`[tool] calculate(${expression}) = ${result}`);
    return result;
  },
});

export async function ask(prompt, options = {}) {
  const session = new LlamaChatSession({
    contextSequence: context.getSequence(),
    autoDisposeSequence: true,
    systemPrompt: options.system,
  });
  const answer = await session.prompt(prompt, {
    temperature: 0,
    maxTokens: options.maxTokens ?? 300,
    grammar: options.grammar,
  });
  session.dispose();

```javascript
    const cleaned = text.replace(/\s+/g, '').replace(/[^0-9+]/g, '');
    
    // التحقق من أن الرقم يحتوي على 11 رقمًا
    if (cleaned.length !== 11 && cleaned.length !== 12) {
        return false;
```

harness.mjs (جزء)
export async function score(prompt, cases) {
  let right = 0;
  for (const [message, expected] of cases) {
    const label = (await ask(`${prompt}\n\n"${message}"`)).toLowerCase();
    right += label === expected ? 1 : 0;
  }
  return right;
}

البرمجة الثنائية مع الذكاء الاصطناعي

في هذا الفصل ستتعلم أن

  • تعمل مع مساعد بخطوات صغيرة: مواصفات، فخطة، فاختبارات، فكود
  • تتحقق أن الاختبارات الجديدة تفشل للسبب الصحيح قبل وجود أي كود
  • ترسل الفشل الحقيقي من جديد، وتعرف متى تتوقف
  • تحصل على كود بأسلوب مشروعك بأن تُري النموذج مشروعك
  • تقرأ ما كتبه المساعد، ومنه شرحه، بعين المراجع
  • تحتفظ بالقرارات، وبالأمثلة التي تعرّف «الصحيح»، لنفسك

في البرمجة الثنائية يتشارك مطوران مهمة واحدة: أحدهما يكتب، والآخر يراقب ويسأل ويفكر في الخطوة التالية، ثم يتبادلان. ومع المساعد، أنت دائماً المطور الثاني، حتى وهو يكتب. وهذا الفصل يبني ميزة صغيرة مع نموذج الكتاب بالخطوات التي يتبعها زوج متأنٍّ، ويشغّل الاختبارات الحقيقية في كل خطوة، ويعرض ما حدث فعلاً، وفي النسخة العربية لم ينجح كل شيء: والفشل علّم أكثر من النجاح.

بِت

في النسخة الإنجليزية من هذا الفصل سار التشغيل الأخير بسلاسة: نجحت الاختبارات الست من أول جولة. وبالعربية، بالنموذج نفسه، تعثّر الزوج طويلاً ولم يصل وحده. والقسم 4.6 يروي ذلك، لأنه الجزء الذي ستقابله في العمل.

4.1 ابدأ بالمواصفات

الميزة صغيرة وحقيقية: تقسيم فاتورة مقهى بين أصدقاء، بقروش صحيحة، مع بقشيش. وقبل أن تطلب أي شيء، اكتب ما معنى «صحيح»، ومعه أمثلة حسبتها بنفسك:

الدالة splitBill(total, people, tipPercent) تقسم فاتورة مقهى. total بالقروش (عدد صحيح)، وpeople عدد صحيح لا يقل عن 1، وtipPercent عدد صحيح مثل 10. البقشيش هو total * tipPercent / 100، مقرّباً لأقرب قرش. وتعيد مصفوفة فيها مبلغ صحيح لكل شخص، مجموعها بالضبط total مع البقشيش؛ وحين لا تنقسم بالتساوي، يدفع الأوائل قرشاً زائداً. وترمي RangeError إن كان people أقل من 1. أمثلة: splitBill(1000, 3, 10) تعيد [367, 367, 366]؛ وsplitBill(995, 2, 15) تعيد [572, 572]؛ وsplitBill(1000, 4, 0) تعيد [250, 250, 250, 250].

الأمثلة أهم ما فيها. «splitBill(1000, 3, 10) تعيد [367, 367, 366]» تحسم أسئلة تتركها الكلمات مفتوحة: هل يُقرَّب البقشيش قبل القسمة أم بعدها، ومن يدفع القرش الزائد، وهل المبالغ أعداد صحيحة. ونسخة مبكرة من هذا الفصل كانت بلا أمثلة، فتوقعت اختبارات النموذج [27, 27, 27, 27] لفاتورة مجموعها 110: واثقة ومحددة وخاطئة. فالنموذج يستطيع كتابة الاختبارات؛ لكنه لا يستطيع أن يعرف ما يعنيه عملك بـ «صحيح». هذه المعرفة لك، والأمثلة مكانها.

هذه الخطوة موجودة لأن أول نسخة من هذا الفصل كانت بدونها. فاختبارات النموذج استوردت assert من node:test، وليس هناك، ففشل كل اختبار بـ «assert.deepEqual is not a function». ثم طلبت حلقتي من النموذج أن يصلح الكود «لا الاختبارات»، فلم يستطع إصلاح مشكلة ليست في الكود، وردّ أخيراً بشرح بدل الكود، فكتبته الحلقة في bill.mjs. وخرج من ذلك درسان، وكلاهما في الكود الآن: الاختبارات تُفحص قبل أن يُوثق بها، والجواب الذي لا كتلة كود فيه لا يُكتب فوق ملف أبداً.

4.4 أرسل الفشل الحقيقي

  if (result.ok || round === 4) {

}

بعد رفض النموذج إصلاح require، أضاف المطور تلميحاً، جملة واحدة تسمّي العُرف: «هذا المشروع يستخدم وحدات ES، فاستخدم import لا require أبداً». فاختفى require، لكن النسخة التالية أسقطت أسطر الاستيراد كلها («test is not defined»). ثم كُتب الكود، وفي إحدى جولات الإصلاح، وقد طُلب منه إصلاح الكود لا الاختبارات، أضاع الكود تصديره لحظة. وبعد أربع جولات توقفت الحلقة. وما تركته: splitBill صحيحة، وملف اختبارات ينقصه سطرا استيراد:

bill/bill.mjs
export const splitBill = (total, people, tipPercent) => {
  if (people < 1) {
    throw new RangeError('people must be at least 1');
  }

  const tip = Math.round(total * tipPercent / 100);
  const totalWithTip = total + tip;

  const base = Math.floor(totalWithTip / people);
  const remainder = totalWithTip % people;

  const result = [];
  for (let i = 0; i < people; i++) {
    result.push(base + (i < remainder ? 1 : 0));
  }

  return result;
};

```js
function calc(o, d) {
  var t = 0;
  for (var i = 0; i < o.l.length; i++) {
    var x = o.l[i];
    t += x.p * x.q;
  }
  if (d && t > 20000) t = t - Math.floor(t * 0.1);
  if (o.c == "VIP") t = t - 500;
  return t < 0 ? 0 : t;
}
```

### السبب الأرجح في أن الدالة تُعيد مشروبات خاطئة:

> **الدالة تُستخدم `menu.find(...)` لاسترجاع الاسم بناءً على السعر، لكنها لا تضمن أن كل سعر مُستخرج من `sort().reverse().slice(0,3)` يُوجد في قائمة المشروبات الأصلية.**  
> أي أن السعر قد يكون مُكررًا، أو أن الدالة تُعيد الاسم من مشروب له سعر مطابق، لكنها لا تُعيد الأسماء المُقابلة للأسعار الأعلى، بل تُعيد القيم المُرتّبة من السعر، مما يؤدي إلى **إرجاع أسماء مشروبات غير مطابقة للأسعار الأعلى**.

مثلاً:
- السعر الأعلى هو 1050 (مُوْحَد لـ "mocha")
- ثم 900 (لـ "cold brew")
- ثم 450 (لـ "latte")

لكن الدالة تُعيد:
```js
menu.find(drink => drink.price === 1050).name → "mocha"
menu.find(drink => drink.price === 900).name → "cold brew"
menu.find(drink => drink.price === 450).name → "latte"
```

لكن المشكلة أن الدالة **تُعيد القيم المُرتّبة من السعر، لكنها لا تُعيد الأسماء المُقابلة للأسعار الأعلى**، بل تُعيد الأسماء من خلال مقارنة السعر فقط، وهذا **مُمكن أن يُعطي أسماء خاطئة إذا كان هناك سعر مكرر**.

لكن في هذا المثال، لا يوجد تكرار، لذا السبب الأرجح هو:

> ✅ **الدالة تُعيد الأسماء بناءً على السعر، لكنها لا تُعيد الأسماء المرتبة حسب السعر الأعلى، بل تُعيد الأسماء من خلال مقارنة السعر فقط، مما يُسبب خللًا في التماثل بين السعر والاسم.**

لكن المشكلة الأكبر هي أن:

> ❌ **الدالة لا تُعيد الأسماء المرتبة حسب السعر الأعلى، بل تُعيد الأسماء من خلال ترتيب السعر، ثم تبحث عن الاسم من القائمة، لكنها لا تضمن أن كل سعر مُستخرج يُوجد في القائمة.**

لكن في هذه الحالة، كل سعر موجود، لذا المشكلة الأعمق هي:

> ❌ **الدالة لا تُعيد الأسماء المرتبة حسب السعر الأعلى، بل تُعيد الأسماء من خلال ترتيب السعر، ثم تبحث عن الاسم من القائمة، لكنها لا تُعيد الأسماء المرتبة حسب السعر الأعلى، بل تُعيد الأسماء من خلال ترتيب السعر، ثم تبحث عن الاسم من القائمة، مما يُسبب خللًا في الترت

(no fixed function in the answer)

لا إصلاح. فقد قرر النموذج أن المشكلة أسعار مكررة وfind، ولا أسعار مكررة هنا، ثم أعاد الجملة نفسها تقريباً مرة بعد مرة، «لا تُعيد الأسماء المرتبة حسب السعر الأعلى، بل تُعيد الأسماء من خلال ترتيب السعر»، حتى نفدت مساحة الجواب، ثمانمئة رمز، قبل أي كود. والسبب الحقيقي .sort() بلا دالة مقارنة، فهي ترتب الأرقام كنصوص: «1050» قبل «250» لأن «1» قبل «2»، فبعد reverse() تصير الموكا الأخيرة.

حين تصلح خطأً، احتفظ بإعادة الإنتاج اختباراً. فهي تفشل بالطريقة الصحيحة أصلاً؛ وبعد الإصلاح تنجح؛ ومن بعدها تحرس ذلك الفشل بعينه. وتتبّع الأخطاء مع مساعد ينتج سكربتات صغيرة كثيرة؛ والتي تعيد إنتاج فشل حقيقي منها تستحق البقاء.

في هذا الفصل ستتعلم أن

7.1 الكود المُختبَر

دالة صغيرة بثلاث قواعد وثلاثة حدود، وأمثلة في تعليقها، كما علّم الفصل 4:

price.mjs
// Price of an order line in piastres: unit price times quantity,
// 10% off from the 10th cup, and never more than 5000 off.
// Examples: linePrice(1000, 9) is 9000; linePrice(1000, 10) is 9000;
// linePrice(6000, 10) is 55000.
export function linePrice(unit, quantity) {
  if (quantity < 1) {
    throw new RangeError("quantity must be at least 1");
  }
  const full = unit * quantity;
  const off = quantity >= 10 ? Math.min(Math.floor(full / 10), 5000) : 0;
  return full - off;
}

7.2 الاختبارات، مقيسة

لمجموعة الاختبارات وظيفتان: أن تنجح حين يكون الكود صحيحاً، وأن تفشل حين يكون خاطئاً. والأولى تُفحص بتشغيل الاختبارات على الكود الحقيقي. والثانية تُفحص بـالمتحوّرات (mutants): نسخ من الكود، في كل منها خطأ صغير مزروع. والمجموعة الجيدة تفشل على كل واحدة؛ وكل متحوّر ينجح خطأ ستتركه اختباراتك يمر.

والاختبارات الثلاثة الصحيحة كشفت أربعة متحوّرات من خمسة. والناجي Math.floor وقد صارت Math.round: لا اختبار يستخدم سعراً لخصمه كسر أكبر من النصف، فيعطي التقريب للأعلى والتقريب للأسفل النتيجة نفسها في كل اختبار. وهذه فجوة حقيقية، وجدها فحص آلي، وهي بالضبط النوع الذي تفوّته المراجعة بالعين. وفي النسخة الإنجليزية، كتب النموذج ثمانية اختبارات، ثلاثة منها خاطئة عند سقف الخصم، وأعاد بعد الإفادة ملفاً لا يُحلَّل أصلاً؛ والستة الصحيحة كشفت أربعة من خمسة، وكان الناجي هو نفسه.

وحارسان جعلا هذه الأرقام صادقة، وأُضيف كلاهما بعد أن كذبت التشغيلات الأولى. ملف الاختبارات الذي يفشل على الكود الحقيقي «يكشف» كل متحوّر، لأنه يفشل في كل مكان؛ فلا تُحسب الدرجة إلا للاختبارات التي تنجح على الكود الحقيقي. والملف بلا اختبارات ينجح: فقد أعاد النموذج في نسخة سابقة، بدل الاختبارات، نسخة من price.mjs، فأعلن البرنامج النجاح بصفر اختبارات. وقد احتاج الإصلاح محاولتين: فعدّ الاختبارات الذي يطبعه المشغّل لم يكفِ، لأن Node يعدّ الملف الفارغ نفسه اختباراً ناجحاً. والآن، الملف الذي ليس فيه test( واحد يفشل قبل أن يُشغَّل.

انتبه

الأخضر يعني «لم يفشل شيء»، لا «اختُبر كل شيء». وقبل أن تثق بمجموعة اختبارات كتبها أي أحد، نموذجاً أو إنساناً، افحص ثلاثة أشياء: أنها شغّلت اختبارات (عُدّها)، وأنها تفشل حين يكون الكود خاطئاً (اكسر الكود عمداً)، وأن قيمها المتوقعة جاءت من المواصفات، لا من الكود الذي تختبره.

اختبر نفسك

  1. لماذا يكشف الاختبار الذي يفشل على الكود الحقيقي كل متحوّر؟
  2. ماذا يحتاج الاختبار ليكشف متحوّر Math.round؟
  3. لماذا الملف الفارغ خطر في خط إنتاج آلي؟
