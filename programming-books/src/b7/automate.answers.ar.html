export function toIso(text) {
  let m;
  if ((m = text.match(/^(\d{4})[-/.](\d{1,2})[-/.](\d{1,2})$/))) return iso(m[1], m[2], m[3]);
  if ((m = text.match(/^(\d{1,2})[-/.](\d{1,2})[-/.](\d{4})$/))) return iso(m[3], m[2], m[1]);
  if ((m = text.match(/^(\d{1,2})\s+(\S+)\s+(\d{4})$/))) return iso(m[3], MONTHS[m[2].slice(0, 3).toLowerCase()] ?? MONTHS[m[2]], m[1]);
  if ((m = text.match(/^(\S+)\s+(\d{1,2}),\s*(\d{4})$/))) return iso(m[3], MONTHS[m[1].slice(0, 3).toLowerCase()], m[2]);
  return null;
}
