  2. localStorage يقرؤه كل سكربت يعمل في الصفحة، فخطأ واحد في حقن السكربتات بين المواقع، أو سكربت طرف ثالث مخترق، يستطيع نسخ الرمز واستعماله من أي مكان حتى تنتهي مدته. احفظ الرمز في كوكي httpOnly وsameSite وsecure، يرسلها المتصفح وحده ولا يقرؤها أي سكربت؛ وتطبيق الويب للجوال موقع لا أكثر ويستعمل الكوكي أيضاً. ولا يحفظ رمز bearer إلا التطبيقات الأصلية، التي لا كوكيز متصفح لها، في المخزن الآمن للمنصة (Keychain وKeystore).
  3. curl -s localhost:3000/owner -H 'cookie: session=anything'. يرى الوكيل كوكي بالاسم الصحيح فيدع الطلب يمر. ثم تنادي الصفحة requireUser، فيبصم anything، ولا يجد جلسة بتلك البصمة، ولا يحصل على مستخدم؛ فتحوّل الصفحة إلى صفحة الدخول دون أن تقرأ أي بيانات. لم يكن الوكيل الفحص قط، فتجاوزه لا يكسب شيئاً.

export async function GET(request: Request) {
  try {
    const user = await requireUser(["owner"], request);
    return Response.json({ data: await customerEmailsFor(user) });
  } catch (error) {
    return authProblem(error);
  }
}
